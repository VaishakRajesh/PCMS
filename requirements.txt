# PortfolioCMS dependencies.
# Install everything with:  pip install -r requirements.txt
#
# What each package is for:
#   fastapi            - the web framework (routes, forms, file uploads)
#   uvicorn            - the server that runs the app (python main.py)
#   python-multipart   - lets FastAPI read HTML forms and uploaded files
#   itsdangerous       - signs the login cookie so users can't fake it
#   jinja2             - HTML templates in the templates/ folder
#   passlib[bcrypt]    - password hashing (never store plain passwords)
#   bcrypt==4.0.1      - PINNED: passlib breaks with bcrypt 4.1+,
#                        so do NOT upgrade this line casually.
fastapi
uvicorn
python-multipart
itsdangerous
jinja2
passlib[bcrypt]
bcrypt==4.0.1
